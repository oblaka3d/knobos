cmake_minimum_required(VERSION 3.16)
project(knobos_host_tests C)
add_library(unity unity/unity.c)
target_include_directories(unity PUBLIC unity)
add_executable(test_carousel test_carousel.c ../../components/ui/carousel.c)
target_include_directories(test_carousel PRIVATE ../../components/ui/include)
target_link_libraries(test_carousel unity)
add_executable(test_ids test_ids.c ../../components/core/ids.c)
target_include_directories(test_ids PRIVATE ../../components/core/include)
target_link_libraries(test_ids unity)
enable_testing()
add_test(NAME carousel COMMAND test_carousel)
add_test(NAME ids COMMAND test_ids)
